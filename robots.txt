# Mockups and the iframe test carry noindex. They must stay crawlable so Google
# can see that tag. Blocking them is "Indexed, though blocked by robots.txt".

# Block archived / superseded pages — duplicate content of live pages, and their
# stale internal links point at dropped URLs (→ 404s in Search Console if crawled)
Disallow: /_archive/

# Retired /fr/ and /es/ URLs are English noindex stubs, or a real 404 once the
# homepage-only indexes were removed. Do not Disallow them. A robots block
# hides noindex from Google and is reported as "Indexed, though blocked by robots.txt".

# Internal review notes, not user-facing content.
Disallow: /reviews/

# Repo notes for maintainers. Not a public page.
Disallow: /_docs/

# Homepage article-card fragment — injected into /, not a standalone page.
# Wrapping it in a noindex document would be inserted into the homepage.
Disallow: /data/article-cards-deferred.html

# /data/*.json and /data/*.csv are public downloads, linked from the homepage
# Dataset schema and /about.html. Blocking them left the URLs indexed without
# a crawl. Leave them crawlable.

# RSS feeds are not linked from pages and are not in the sitemap. GitHub Pages
# cannot send a noindex header for XML, so these stay out of the crawl.
Disallow: /feed.xml
Disallow: /feed-breakthrough.xml
Disallow: /feed-guide.xml
Disallow: /feed-kids.xml
Disallow: /feed-reality-check.xml
Disallow: /feed-safety-alert.xml

# hub. Query URLs such as /supplement.html?slug= still render long-tail records
# that have no prerendered page. Do not Disallow the shells: Google cannot see
# noindex on a blocked URL.
